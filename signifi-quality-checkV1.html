Supabase Schema

Run these in your Supabase SQL Editor to set up the tables.

CREATE TABLE deployments ( id uuid PRIMARY KEY DEFAULT gen_random_uuid(), task_id text UNIQUE, parent_ticket text, display_name text, client text, kiosk_id text, location text, stage text, activity_type text, tech text, week_no integer, install_date date, uploaded_at timestamptz DEFAULT now() );
CREATE TABLE quality_scores ( id uuid PRIMARY KEY DEFAULT gen_random_uuid(), deployment_id uuid REFERENCES deployments(id) ON DELETE CASCADE, role text CHECK (role IN ('pm','logistics','install')), track text CHECK (track IN ('new','revisit')), submitted_by text, accountability text, comments text, submitted_at timestamptz DEFAULT now(), answers jsonb, score numeric(5,2), UNIQUE (deployment_id, role) );
ALTER TABLE deployments ENABLE ROW LEVEL SECURITY; ALTER TABLE quality_scores ENABLE ROW LEVEL SECURITY; CREATE POLICY "allow all" ON deployments FOR ALL USING (true) WITH CHECK (true); CREATE POLICY "allow all" ON quality_scores FOR ALL USING (true) WITH CHECK (true);
